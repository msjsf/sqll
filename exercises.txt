CREATE DATABASE IF NO EXISTS sales;

USE Sales;

/*create tables*/
CREATE TABLE Customers(
    customer_id int AUTO_INCREMENT,
    first_name varchar(255) NUT NULL,
    email_address VARCHAR(255) not NULL,
    number_of_complaints INT DEFAULT 0,
    PRIMARY KEY (customer_id),
    UNIQUE KEY (email_address)
);
CREATE TABLE Companies(
    company_id INT AUTO_INCREMENT,
    company_name VARCHAR(255) NOT NULL,
    headquarters_phone_number varchar(100),
    PRIMARY KEY(company_id)
);

CREATE TABLE Items(
    item_code VARCHAR(10) UNIQUE NOT NULL,
    item VARCHAR(255) NOT NULL,
    unit_price DECIMAL(5,2) NOT NULL,
    company_id int,
    PRIMARY KEY(item_code),
    FOREIGN KEY(company_id) REFERENCES Companies(company_id) ON DELETE CASCADE
);

CREATE TABLE Sales(
    purchase_number INT AUTO_INCREMENT,
    date_of_purchase DATE NOT NULL,
    customer_id INT,
    item_code VARCHAR(10) NOT NULL,
    PRIMARY KEY (purchase_number),
    FOREIGN KEY(customer_id) REFERENCES Customers(customer_id) ON DELETE CASCADE,
    FOREIGN KEY (item_code) REFERENCES Items(item_code) ON DELETE CASCADE


);

mostrar todos los productos con su empresa

SELECT Items.item, Companies.company_name
from Items
JOIN Companies on Items.company_id=Companies.company_id



mostrar compras con nombre del cliente

select Sales.purchase_number,
    Customers.first_name,
    Customers.last_name,
    Items.item,
    Sales.date_of_purchase
from Sales
JOIN Customers ON Sales.customer_id=Customers.customer_id
JOIN Items ON Sales.Items.item_code=Items.item_code
